numpy

# Needed by gui_web/frame_server.py to JPEG-encode camera frames for the
# MJPEG stream the web UI displays.
Pillow

# Qt6 + QtWebEngine: hosts demo_ui/ in a QWebEngineView. Must be the full
# PySide6 metapackage (QtWebEngine ships in PySide6-Addons), not
# PySide6-Essentials. PyQt6 has no prebuilt aarch64 wheel and fails to build
# from source without qmake/Qt6 dev headers on Raspberry Pi OS
# bookworm/trixie, so we use PySide6 (same Qt6, official prebuilt aarch64
# wheels via piwheels/PyPI).
PySide6

# Relay (hardware/relay_api.py, config.RUN_WITH_RELAY = True)
lgpio

# Card reader (default backend: config.CARD_READER_BACKEND = "gwiot_hid" --
# card_backends_impl/gwiot_hid_card_reader.py reads the GWIOT USB HID
# keyboard-emulation reader via evdev).
evdev
